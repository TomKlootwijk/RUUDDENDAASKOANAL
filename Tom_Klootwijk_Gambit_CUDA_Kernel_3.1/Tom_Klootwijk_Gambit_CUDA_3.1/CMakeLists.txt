cmake_minimum_required(VERSION 3.24)
project(TomKlootwijkGambit VERSION 3.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
option(GAMBIT_ENABLE_CUDA "Build native CUDA sm_120 executable (requires CUDA 12.8+)" ON)
add_library(gambit_host STATIC src/host.cpp)
target_include_directories(gambit_host PUBLIC include)
if(MSVC)
  target_compile_options(gambit_host PRIVATE /W4)
else()
  target_compile_options(gambit_host PRIVATE -Wall -Wextra -Wpedantic)
endif()
add_executable(gambit_cpu src/cpu_main.cpp)
target_link_libraries(gambit_cpu PRIVATE gambit_host)
add_executable(gambit_unit tests/test_model.cpp)
target_link_libraries(gambit_unit PRIVATE gambit_host)
enable_testing()
add_test(NAME integer_contract COMMAND gambit_unit)
if(GAMBIT_ENABLE_CUDA)
  include(CheckLanguage)
  check_language(CUDA)
  if(NOT CMAKE_CUDA_COMPILER)
    message(FATAL_ERROR "CUDA compiler absent. Install CUDA 12.8+ for GPU build. Explicit -DGAMBIT_ENABLE_CUDA=OFF builds CPU only, NOT a GPU verification.")
  endif()
  # Numeric architectures are understood by nvcc 12.8+. Do not substitute sm_100.
  set(CMAKE_CUDA_ARCHITECTURES "120-real;120-virtual" CACHE STRING "GPU code target")
  enable_language(CUDA)
  if(CMAKE_CUDA_COMPILER_VERSION VERSION_LESS 12.8)
    message(FATAL_ERROR "Native sm_120 compilation requires CUDA Toolkit 12.8 or newer")
  endif()
  add_executable(gambit_cuda src/gambit_cuda.cu)
  target_link_libraries(gambit_cuda PRIVATE gambit_host)
  set_target_properties(gambit_cuda PROPERTIES CUDA_STANDARD 17 CUDA_STANDARD_REQUIRED ON CUDA_RUNTIME_LIBRARY Static)
  target_compile_options(gambit_cuda PRIVATE "$<$<COMPILE_LANGUAGE:CUDA>:-lineinfo;--fmad=false;-Xptxas=-v>")
endif()
