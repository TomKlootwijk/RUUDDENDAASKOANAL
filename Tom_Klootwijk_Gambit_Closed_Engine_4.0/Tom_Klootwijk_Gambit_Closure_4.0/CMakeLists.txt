cmake_minimum_required(VERSION 3.24)
project(TomKlootwijkGambitClosure VERSION 4.0.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
option(GAMBIT_ENABLE_CUDA "Build optional, locally unverified CUDA backend" OFF)
option(GAMBIT_SANITIZE "Address and undefined-behavior instrumentation (GCC/Clang)" OFF)
add_library(closure_core src/legacy_host.cpp src/engine.cpp src/boolean.cpp src/capsule.cpp src/run.cpp)
target_include_directories(closure_core PUBLIC include)
if(MSVC)
  target_compile_options(closure_core PRIVATE /W4)
else()
  target_compile_options(closure_core PRIVATE -Wall -Wextra -Wpedantic)
  if(GAMBIT_SANITIZE)
    target_compile_options(closure_core PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(closure_core PUBLIC -fsanitize=address,undefined)
  endif()
endif()
add_executable(gambit src/main.cpp)
target_link_libraries(gambit PRIVATE closure_core)
add_executable(gambit_tests tests/native_test.cpp)
target_link_libraries(gambit_tests PRIVATE closure_core)
enable_testing()
add_test(NAME closure_native_contract COMMAND gambit_tests ${CMAKE_CURRENT_SOURCE_DIR}/assets/micro_xy.gblut)
if(GAMBIT_ENABLE_CUDA)
  include(CheckLanguage)
  check_language(CUDA)
  if(NOT CMAKE_CUDA_COMPILER)
    message(FATAL_ERROR "BLOCKED: CUDA compiler absent; CPU testing is not GPU verification")
  endif()
  set(CMAKE_CUDA_ARCHITECTURES "120-real;120-virtual" CACHE STRING "Requested RTX 5070 Ti Laptop target")
  enable_language(CUDA)
  if(CMAKE_CUDA_COMPILER_VERSION VERSION_LESS 12.8)
    message(FATAL_ERROR "sm_120 requires CUDA 12.8 or newer")
  endif()
  add_executable(gambit_cuda src/cuda.cu)
  target_link_libraries(gambit_cuda PRIVATE closure_core)
  set_target_properties(gambit_cuda PROPERTIES CUDA_STANDARD 17 CUDA_STANDARD_REQUIRED ON)
  target_compile_options(gambit_cuda PRIVATE "$<$<COMPILE_LANGUAGE:CUDA>:-lineinfo;--fmad=false;-Xptxas=-v>")
endif()
